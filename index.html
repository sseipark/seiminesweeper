function startBotAI() { clearInterval(botInterval); startFriendTimer(); // 봇 클릭 속도 완화 (1.8초 ~ 2.5초 간격 랜덤) const getHumanDelay = () => Math.floor(Math.random() * 700) + 1800; const botStep = () => { if (gameOver || !isBotActive) return; // 봇의 실수 확률 (12% 확률로 지뢰 밟고 펑) const isError = Math.random() < 0.12; if (isError) { applyFriendAction({ action: 'boom' }); triggerGameOver(true, '🎉 봇이 지뢰를 밟고 폭발했습니다! 나의 승리!'); return; } // 인간미 있는 느릿한 진행률 상승 let currentPct = parseInt(friendProgressStat.textContent.replace(/[^0-9]/g, '')) || 0; let nextPct = Math.min(100, currentPct + Math.floor(Math.random() * 8) + 3); friendProgressStat.textContent = `진행률: ${nextPct}%`; if (nextPct >= 100) { friendFaceBtn.textContent = '😎'; triggerGameOver(false, '😭 봇이 먼저 진행률 100%를 달성했습니다! 패배!'); return; } botInterval = setTimeout(botStep, getHumanDelay()); }; botInterval = setTimeout(botStep, getHumanDelay()); }
